fn main() {
    let src_dir = std::path::Path::new("src");

    // Fail the build when src/parser.c is stale relative to grammar.js.
    // CI additionally runs `tree-sitter generate && git diff --exit-code`.
    let grammar = Path::new("../tree-sitter-specforge/grammar.js");
    let parser = src_dir.join("parser.c");
    let stale = mtime_of(grammar) > mtime_of(&parser);
    assert!(
        !stale,
        "src/parser.c is older than grammar.js — run `tree-sitter generate`"
    );

    let mut c_config = cc::Build::new();
    c_config.std("c11").include(src_dir);
    c_config.file(&parser);
    println!("cargo:rerun-if-changed={}", parser.to_str().unwrap());
    println!("cargo:rerun-if-changed=grammar.js");
    c_config.compile("tree-sitter-specforge");
}

resolve.rs implements the 5-step import cascade (relative, @-aliases, extension stubs, bare-from-spec_root) with a shared index-fallback rule (path.spec beats path/index.spec), plus a normalized-path containment check that rejects ../ traversal outside spec_root. resolution.rs tests (27.5KB) cover barrels, selective re-export misses (W027), and alias edge cases; linker tests pin W063 and E003 suggestion behavior. The traversal guard in try_resolve_relative is exactly the defensive touch this kind of path plumbing needs.

exemplar — the good pattern
    // Step 1: Relative paths
    if import_path.starts_with("./") || import_path.starts_with("../") {
        return try_resolve_relative(spec_root, importing_file, import_path);
    }

    // Step 2: @-prefixed paths (aliases or extension stubs)
    if let Some(rest) = import_path.strip_prefix('@') {
        // Try alias first
        if let Some(result) = try_resolve_alias(spec_root, rest, config) {
            return result;
        }
        // Fall through to extension stub
        return try_resolve_extension(rest);
    }

    // Step 3: Bare paths — resolve from spec_root
    try_resolve_bare(spec_root, import_path)
}
