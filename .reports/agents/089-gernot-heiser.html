#[cfg(test)]
mod dual_source_parity {
    use super::*;

    const RULES_JSON: &[u8] = include_bytes!(
        "../../../../extensions/formal/src/describe_validation_rules.json");

    /// The wasm describe payload and the native rule table MUST agree.
    /// Guards the hand-synced dual copy from silent drift.
    #[test]
    fn native_rules_match_embedded_json() {
        let json_rules: Vec<ValidationRuleDescriptor> = serde_json::from_slice(
            &serde_json::from_slice::<serde_json::Value>(RULES_JSON)
                .unwrap()["items"].to_string().into_bytes())
            .unwrap();
        let native = FormalExtension.validation_rules();
        assert_eq!(native, json_rules, "formal rule drift between JSON and Rust");
    }
}
