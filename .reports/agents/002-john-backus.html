/// Regenerate docs/extension-inventory.md from the actual extension
/// manifests, so the document can never describe a nonexistent extension.
fn gen_extension_inventory() -> anyhow::Result<()> {
    let mut rows = Vec::new();
    for entry in std::fs::read_dir("extensions")? {
        let dir = entry?.path();
        let manifest: serde_json::Value = serde_json::from_str(&std::fs::read_to_string(
            dir.join("handshake.json"),
        )?)?;
        let name = manifest["name"].as_str().unwrap_or_default();
        let kinds = manifest["kinds"].as_array().map_or(0, |k| k.len());
        let edges = manifest["edges"].as_array().map_or(0, |e| e.len());
        let deps: Vec<&str> = manifest["depends_on"]
            .as_array()
            .map(|d| d.iter().filter_map(|v| v.as_str()).collect())
            .unwrap_or_default();
        rows.push(format!("| {name} | {kinds} | {edges} | {} |", deps.join(", ")));
    }
    rows.sort();
    let doc = format!(
        "# Extension Inventory\n\nGenerated by `cargo xtask gen-extension-inventory`. Do not edit by hand.\n\n| Extension | Kinds | Edges | Depends on |\n|---|---|---|---|\n{}\n",
        rows.join("\n")
    );
    std::fs::write("docs/extension-inventory.md", doc)?;
    Ok(())
}
