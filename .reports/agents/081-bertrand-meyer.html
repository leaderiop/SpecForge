/// F030-formal: X/not_X contradiction lint over requires names,
/// per fa_pattern_e030_contradiction pattern 1.
fn check_contradictory_requires(behavior: &Node) -> Vec<Diagnostic> {
    let mut diags = Vec::new();
    let names: Vec<&str> = requires_names(behavior).collect();
    for (i, a) in names.iter().enumerate() {
        for b in &names[i + 1..] {
            let negated =
                a.strip_prefix("not_").is_some_and(|r| r == *b)
                    || b.strip_prefix("not_").is_some_and(|r| r == *a);
            if negated {
                diags.push(Diagnostic {
                    code: "F030".into(), severity: Severity::Error,
                    message: format!("behavior '{}': contradictory requires '{a}' vs '{b}'",
                        behavior.id.raw),
                    span: Some(behavior.source_span.clone()), suggestion: None,
                });
            }
        }
    }
    diags
}

//! Owned diagnostic code registry.
//! E0xx: compiler codes (unchanged). W0xx: warnings.
//! F0xx: formal-analysis codes owned by @specforge/formal.

pub const F030_CONTRADICTION: &str = "F030";
pub const F031_LAYERING_LOSS: &str = "F031";
pub const F032_LAYERING_CYCLE: &str = "F032";
pub const F033_EVENT_GRAPH_CYCLE: &str = "F033";
pub const F034_INTEGRITY: &str = "F034";

/// Single lookup behind specforge explain so meanings can never
/// diverge between spec prose and emitted diagnostics again.
pub fn explain(code: &str) -> Option<&'static str> {
    match code {
        F030_CONTRADICTION => Some("Structured condition contradiction"),
        F031_LAYERING_LOSS => Some("Refinement weakened an abstract ensures"),
        F032_LAYERING_CYCLE => Some("Refinement chain cycle"),
        F033_EVENT_GRAPH_CYCLE => Some("Unmitigated event-graph cycle"),
        F034_INTEGRITY => Some("Wasm module integrity check failed"),
        _ => None,
    }
}
