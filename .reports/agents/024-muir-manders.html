after — proposed fix
        // Context-aware keywords anywhere on the line, prefix-scored.
        let line_prefix = word_at_position(&doc.content(), pos).to_lowercase();
        let kind_reg = state.kind_registry();
        let mut kw_items: Vec<CompletionItem> = kind_reg
            .keywords()
            .filter(|kw| kw.starts_with(&line_prefix) || jaro_winkler(kw, &line_prefix) > 0.7)
            .map(|kw| CompletionItem {
                label: kw.clone(),
                kind: Some(CompletionItemKind::KEYWORD),
                sort_text: Some(rank(kw, &line_prefix)),
                ..Default::default()
            })
            .collect();
        items.append(&mut kw_items);
