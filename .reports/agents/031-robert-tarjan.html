/// Single cycle primitive: Tarjan SCC. Only multi-member SCCs (or
/// self-loops) are returned; W061, W003, LSP and emitter rules consume this.
pub fn strongly_connected_components(&self) -> Vec<Vec<Sym>> {
    let edges_from: HashMap<Sym, Vec<Sym>> = self.edges.iter().fold(HashMap::new(), |mut m, e| {
        m.entry(e.source).or_default().push(e.target); m
    });
    let mut idx = 0usize; let mut indices = HashMap::new();
    let mut low = HashMap::new(); let mut stack = Vec::new();
    let mut on_stack = HashSet::new(); let mut out = Vec::new();
    fn sc(v: Sym, ef: &HashMap<Sym, Vec<Sym>>, idx: &mut usize,
          ind: &mut HashMap<Sym, usize>, low: &mut HashMap<Sym, usize>,
          st: &mut Vec<Sym>, on: &mut HashSet<Sym>, out: &mut Vec<Vec<Sym>>) {
        ind.insert(v, *idx); low.insert(v, *idx); *idx += 1;
        st.push(v); on.insert(v);
        if let Some(ss) = ef.get(&v) { for &w in ss {
            if !ind.contains_key(&w) { sc(w, ef, idx, ind, low, st, on, out);
                low.insert(v, low[&v].min(low[&w])); }
            else if on.contains(&w) { low.insert(v, low[&v].min(ind[&w])); }
        }}
        if low[&v] == ind[&v] { let mut s = Vec::new();
            while let Some(w) = st.pop() { on.remove(&w); s.push(w); if w == v { break; } }
            if s.len() > 1 || ef.get(&v).is_some_and(|x| x.contains(&v)) { s.sort(); out.push(s); } }
    }
    let mut all: Vec<Sym> = self.nodes.keys().copied().collect(); all.sort();
    for v in all { if !indices.contains_key(&v) { sc(v, &edges_from, &mut idx, &mut indices, &mut low, &mut stack, &mut on_stack, &mut out); } }
    out
}

Recursive DFS in all cycle detectors; graph.rs reports duplicate cycles for parallel edges

mediumC5-02
031 Robert Tarjaneffort Smoves: Cycle-detection implementations

Graph::detect_cycles recurses per successor with no depth bound (stack-overflow risk on deep chains) and pushes one cycle per gray-target hit, so parallel edges duplicate the same loop in W061. The Tarjan primitive reports each SCC once with canonical sorted membership; derive one lexically-smallest rotation per SCC for messages.

/// Canonical, deduplicated cycle paths derived from SCC membership:
/// one lexically-smallest rotation per SCC, so parallel edges and
/// different DFS entry points cannot duplicate a cycle.
pub fn detect_cycles(&self) -> Vec<Vec<Sym>> {
    let mut cycles: Vec<Vec<Sym>> = Vec::new();
    for scc in self.strongly_connected_components() {
        if scc.len() == 1 {
            cycles.push(vec![scc[0], scc[0]]); // self-loop
            continue;
        }
        let member: HashSet<Sym> = scc.iter().copied().collect();
        let mut cycle = vec![scc[0]];
        let mut cur = scc[0];
        while cycle.len() < scc.len() {
            let next = self.edges_from(cur.as_str()).iter()
                .map(|e| e.target)
                .filter(|t| member.contains(t))
                .min()
                .expect("SCC member has an in-SCC successor");
            cycle.push(next);
            cur = next;
        }
        cycles.push(cycle);
    }
    cycles
}
