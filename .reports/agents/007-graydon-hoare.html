### Reserved Words

All entity keywords from installed extensions are reserved identifiers. Using a reserved word as an entity name produces `E013`. The structural keywords `spec` and `ref` are always reserved. Extension-declared keywords (`behavior`, `invariant`, etc.) are reserved when the extension is installed.

//! Reserved-word validation pass (E013, per docs/entity-model.md).
use specforge_common::{Diagnostic, Severity};
use crate::KeywordIndex;

const ALWAYS_RESERVED: &[&str] = &["spec", "ref", "use", "define"];

pub fn check_reserved_words(
    entities: &[Entity],
    keywords: &KeywordIndex,
    diags: &mut Vec<Diagnostic>,
) {
    for entity in entities {
        let kind = entity.kind.raw.as_str();
        if !(ALWAYS_RESERVED.contains(&kind) || keywords.installed().any(|k| k == kind)) {
            continue;
        }
        let name = &entity.name;
        let suggestion = keywords
            .all()
            .min_by_key(|k| edit_distance(k, name))
            .map(|k| format!("`{k}` is a reserved entity keyword; rename this entity"));
        diags.push(Diagnostic {
            code: "E013".into(),
            severity: Severity::Error,
            message: format!("entity name '{name}' collides with reserved keyword '{kind}'"),
            span: Some(entity.span.clone()),
            suggestion,
        });
    }
}
