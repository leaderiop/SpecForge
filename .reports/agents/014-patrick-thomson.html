pub fn link_references(project: &ResolvedProject) -> (Vec<PendingEdge>, Vec<Diagnostic>) {
    let mut edges = Vec::new();
    let mut diagnostics = Vec::new();

    // Global index for close-match suggestions and out-of-import warnings.
    let mut entity_ids: HashMap<Sym, ()> = HashMap::new();
    let mut id_first_seen: HashMap<Sym, (&str, Sym)> = HashMap::new();
    for file in &project.files {
        for entity in &file.spec_file.entities {
            // ... W063 duplicate detection unchanged ...
            entity_ids.insert(entity.id.raw, ());
        }
    }

    // Per-file visibility: declared + exported sets of import targets.
    // compute_file_scopes already builds declared/exported incl. pub-use.
    let visible: HashMap<&str, HashSet<Sym>> = project
        .files
        .iter()
        .map(|file| {
            let mut set: HashSet<Sym> = file
                .spec_file
                .entities
                .iter()
                .map(|e| e.id.raw)
                .collect();
            for import in &file.imports {
                if let Some(scope) = scopes.get(import.path.as_str()) {
                    set.extend(scope.exported.iter().map(|s| Sym::new(s)));
                }
            }
            (file.path.as_str(), set)
        })
        .collect();

    for file in &project.files {
        let visible = &visible[file.path.as_str()];
        for entity in &file.spec_file.entities {
            for entry in entity.fields.entries() {
                if let FieldValue::ReferenceList(refs) = &entry.value {
                    for target_id in refs {
                        let target_sym = Sym::new(target_id);
                        if visible.contains(&target_sym) {
                            edges.push(PendingEdge { source: entity.id.raw, target: target_sym, label: entry.key });
                        } else if entity_ids.contains_key(&target_sym) {
                            // defined somewhere, but not visible here
                            diagnostics.push(warn_not_imported(entity, target_id, entry.key));
                        } else {
                            diagnostics.push(unresolved_e003(entity, target_id, entry.key, &entity_ids));
                        }
                    }
                }
            }
        }
    }
    (edges, diagnostics)
}

E003 span is whole entity, kind-blind; W063 only same-kind cross-file

mediumC3-07
014 Patrick Thomsoneffort Mmoves: Scope-aware resolution
Evidencelinker.rs:17-46,60-99 — span=entity.span; lexical suggestions; PendingEdge has no kind; cross-kind dup ids silent

Every E003 in linker.rs carries span: entity.span.clone() — the whole entity_block — so an LSP squiggle covers dozens of lines instead of the offending identifier token. PendingEdge and the dup-detection logic are kind-blind: PendingEdge { source, target, label } carries no kind, and id_first_seen only warns (W063) when first_kind == entity.kind.raw, so a `behavior x` in one file and an invariant x in another silently coexist in the index and cross-kind references resolve to whichever entity the graph builder picked. Keeping per-token spans is cheap: the parser already walks reference entries and could retain each identifier node's SourceSpan (parse_list builds item spans for other field kinds). Carrying kind on PendingEdge and in the id index would let the linker warn on cross-kind collisions instead of silently merging them.
