//! xtask: repo automation, including the Wasm extension bootstrap.

use std::process::Command;

const EXTENSIONS: &[&str] = &["product", "software", "governance", "formal"];
const TARGET: &str = "wasm32-unknown-unknown";

/// Build every builtin extension so `include_bytes!` in
/// specforge-extism/src/builtins.rs resolves on a fresh clone.
/// Wired into CI and invoked by `specforge doctor --fix`.
fn bootstrap_extensions() -> anyhow::Result<()> {
    for ext in EXTENSIONS {
        let status = Command::new("cargo")
            .args([
                "build", "--release", "--target", TARGET,
                "--manifest-path", &format!("extensions/{ext}/Cargo.toml"),
            ])
            .status()?;
        if !status.success() {
            anyhow::bail!("failed to build extension {ext} for {TARGET}");
        }
    }
    Ok(())
}

fn main() -> anyhow::Result<()> {
    match std::env::args().nth(1).as_deref() {
        Some("bootstrap-extensions") => bootstrap_extensions(),
        _ => eprintln!("usage: xtask bootstrap-extensions"),
    }
    Ok(())
}
