        .relationships
        .iter()
        .filter(|r| keep.contains(r.source.as_str()) && keep.contains(r.target.as_str()))
        .cloned()
        .collect();

    // Filter edge_type_owners to only edges whose declaring extension

        .edge_type_owners
        .iter()
        .filter(|(_, ext)| surviving_extensions.contains(ext.as_str()))
