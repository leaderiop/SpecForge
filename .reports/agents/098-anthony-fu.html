pub struct CoverageArgs {
    /// Minimum fraction of verify statements that must be covered (0.0-1.0)
    #[arg(long)]
    min: f64,

    /// Path to the specforge report directory
    #[arg(long, default_value = "target/specforge")]
    dir: PathBuf,
}

pub fn run(args: &CoverageArgs) -> i32 {
    let Some(graph) = coverage::load_graph(&args.dir.join("graph.json")) else {
        eprintln!("error: graph.json not found; run cargo build first");
        return 2;
    };
    let entries = load_entries(&args.dir);
    let diffs = coverage::compute_coverage_diff(&graph, &entries);
    coverage::format_coverage_summary(&mut std::io::stderr(), &diffs, &graph.timestamp).ok();

    let covered: usize = diffs.iter().map(|d| d.covered).sum();
    let expected: usize = diffs.iter().map(|d| d.expected).sum();
    let ratio = covered as f64 / expected.max(1) as f64;

    if ratio < args.min {
        eprintln!("coverage gate failed: {ratio:.2} < {:.2}", args.min);
        1
    } else {
        0
    }
}
