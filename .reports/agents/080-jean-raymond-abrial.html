/// E031-formal: ensures set inclusion on RefinementRefinesAbstract edges.
/// Named-condition set difference only — no solver, per pattern-catalog.spec.
fn check_ensures_superset(
    graph: &Graph,
    edge_label_to_field: &HashMap<String, String>,
) -> Vec<Diagnostic> {
    let label = edge_label_to_field.get("RefinementRefinesAbstract")
        .map(String::as_str).unwrap_or("RefinementRefinesAbstract");
    let mut diags = Vec::new();
    for edge in graph.edges().iter().filter(|e| e.label == label) {
        let (Some(abs), Some(con)) =
            (graph.node(&edge.source), graph.node(&edge.target)) else { continue };
        let missing: Vec<_> = ensures_names(abs)
            .filter(|c| !ensures_names(con).contains(c))
            .collect();
        if !missing.is_empty() {
            diags.push(Diagnostic {
                code: "F031".into(), severity: Severity::Error,
                message: format!("refinement '{}': concrete ensures missing {}", con.id.raw, missing.join(", ")),
                span: Some(con.source_span.clone()), suggestion: None,
            });
        }
    }
    diags
}

A refinement promises the concrete plan keeps every abstract guarantee. The tool checks the plans are linked and not circular but never compares guarantee lists - like approving a renovation because paperwork links it to the blueprint.

Proposed solution. Implement E031 as a graph pass comparing structured ensures names on RefinementRefinesAbstract edges; set inclusion, no solver needed.
