  // Ask the server for kind metadata instead of hardcoding:
  const kinds: Array<{ name: string; icon?: string }> =
    (await client.sendRequest("specforge/kindRegistry") as any).kinds;

  const kindIcons = Object.fromEntries(
    kinds.map(k => [k.name, k.icon ?? defaultIconFor(k.name)])
  );
