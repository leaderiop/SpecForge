pub(crate) fn on_exit() {
    let entries = registry::drain();
    if entries.is_empty() {
        return;
    }

    let fs = ports::RealFs;
    let binary_name = binary_name();
    let dir = report_dir();

    let json = serde_json::to_string_pretty(&report::build_ref("1.0", &binary_name, &entries))
        .expect("serialize report");
    if let Err(e) = ports::ReportWriter::write_report(&fs, &dir.join(format!("{binary_name}.json")), &json) {
        eprintln!("[specforge-test] failed to write report: {e}");
    }

    // Coverage summary goes through GraphReader, so tests can inject a fake graph.
    let graph_path = dir.join("graph.json");
    if let Some(graph) = ports::GraphReader::read_graph(&fs, &graph_path) {
        let diffs = coverage::compute_coverage_diff(&graph, &entries);
        let _ = coverage::format_coverage_summary(&mut std::io::stderr(), &diffs, &graph.timestamp);
    }
}
