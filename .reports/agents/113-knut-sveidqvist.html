pub fn sanitize_mermaid_text(s: &str) -> String {
    s.replace('"', "'")
        .replace('\n', " ")
        .replace('\r', "")
}

pub fn escape_md_cell(s: &str) -> String {
    s.replace('|', "\\|").replace('\n', " ")
}

#[test]
fn quotes_never_break() {
    assert_eq!(sanitize_mermaid_text("say \"hi\""), "say 'hi'");
}
