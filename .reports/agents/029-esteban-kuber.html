fn diagnostic_to_lsp(diag: &specforge_common::Diagnostic) -> Diagnostic {
    let mut d = Diagnostic {
        range: diag.span.as_ref().map(source_span_to_range).unwrap_or_default(),
        severity: Some(severity_of(diag.severity)),
        code: Some(NumberOrString::String(diag.code.clone())),
        source: Some("specforge".into()),
        message: diag.message.clone(),
        ..Default::default()
    };
    // Surface the resolver's did-you-mean as structured data.
    if let Some(sugg) = &diag.suggestion {
        d.data = Some(serde_json::json!({ "suggestedName": sugg }));
        d.message.push_str(&format!(" Did you mean `{sugg}`?"));
    }
    d
}
