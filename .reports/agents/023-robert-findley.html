        let mut count = 0;

after — proposed fix
    async fn initialized(&self, _: InitializedParams) {
        let roots: Vec<String> = self.root_dir.lock().await.clone().into_iter().collect();
        let this = Arc::new(self.clone());
        tokio::spawn(async move {
            for root in roots {
                this.client.send_notification::<WorkDoneProgressBegin>(
                    WorkDoneProgressBegin { title: "Indexing SpecForge files".into(), ..Default::default() });
                let count = tokio::task::spawn_blocking(move || {
                    this.index_workspace_blocking(&root)
                }).await.unwrap_or(0);
                this.client.log_message(MessageType::INFO,
                    format!("indexed {count} files from {root}")).await;
            }
        });
    }
