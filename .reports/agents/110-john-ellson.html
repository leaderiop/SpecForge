#[derive(Debug, Clone, Default)]
pub struct DotOptions {
    pub cluster_by_extension: bool,
    pub show_labels: bool,
    pub extension_filter: Option<String>,
}

pub fn emit_dot_with(graph: &Graph, options: &DotOptions) -> String {
    let mut out = String::new();
    writeln!(out, "digraph specforge {{").unwrap();
    writeln!(out, "  rankdir=LR;").unwrap();
    if options.cluster_by_extension {
        for (ext, nodes) in group_by_extension(graph) {
            let id = sanitize_cluster_id(&ext);
            writeln!(out, "  subgraph cluster_{id} {{").unwrap();
            writeln!(out, "    label={id}; style=dashed;").unwrap();
            for n in nodes { emit_node(&mut out, n, options); }
            writeln!(out, "  }}").unwrap();
        }
    }

emit_dot sorts edges by (source, target, label) before emission, so identical graph inputs always produce byte-identical DOT. This makes diffs reviewable: the one box that changed is the one line that changed.
